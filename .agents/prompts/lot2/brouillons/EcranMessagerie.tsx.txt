"use client";

import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { useTranslations } from "next-intl";
import { useState } from "react";
import type { FormEvent } from "react";
import { toast } from "sonner";

import { Alerte, Bouton, Champ, EtatChargement, EtatErreur } from "@/components/ui";
import { Label } from "@/components/ui/label";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { peutParametrerPlateforme } from "@/features/administration";
import type {
  ChiffrementMessagerie,
  ReglageMessagerie,
  ResultatDiagnosticMessagerie,
} from "@/features/administration";
import {
  enregistrerMessagerie,
  lireMessagerie,
  reinitialiserMessagerie,
  testerMessagerie,
} from "@/features/administration/adaptateur";
import { ErreurApi } from "@/lib/api";
import { ABSENT } from "@/lib/format";

import { useAdministrateur } from "../../ContexteAdministrateur";
import { CLES_ADMINISTRATION } from "../../cles";
import { CARTE, CARTE_TITRE, LIGNE_DEFINITION, LIGNE_LIBELLE, LIGNE_VALEUR } from "../../tons";

const CHIFFREMENTS: ChiffrementMessagerie[] = ["STARTTLS", "SSL", "AUCUN"];

/** Les réglages d'un compte Gmail avec mot de passe d'application. */
const GMAIL = { hote: "smtp.gmail.com", port: "587", chiffrement: "STARTTLS" as ChiffrementMessagerie };

const PORT_MAX = 65535;

function Ligne({ libelle, valeur }: { libelle: string; valeur: string }) {
  return (
    <div className={LIGNE_DEFINITION}>
      <span className={LIGNE_LIBELLE}>{libelle}</span>
      <span className={LIGNE_VALEUR}>{valeur}</span>
    </div>
  );
}

/**
 * La messagerie de la plateforme : le serveur SMTP, et le diagnostic d'envoi.
 *
 * Le mot de passe d'application n'est jamais renvoyé par le serveur : le champ reste
 * vide, et le laisser vide conserve celui qui est enregistré. Le réglage saisi ici
 * l'emporte sur le `.env` du serveur, qui reste le repli.
 */
export function EcranMessagerie() {
  const requete = useQuery({
    queryKey: CLES_ADMINISTRATION.messagerie(),
    queryFn: lireMessagerie,
  });
  const t = useTranslations("administration");

  if (requete.isPending) return <EtatChargement />;
  if (requete.isError) {
    return (
      <EtatErreur message={t("erreurs.chargement")} onReessayer={() => void requete.refetch()} />
    );
  }

  // `key` : le formulaire repart des valeurs du serveur à chaque enregistrement, sans effet de bord.
  return (
    <div className="flex flex-col gap-4">
      <ReglageActuel reglage={requete.data} />
      <FormulaireMessagerie
        key={requete.data.saisi.modifieLe?.toISOString() ?? "vide"}
        reglage={requete.data}
      />
      <TestMessagerie />
    </div>
  );
}

function ReglageActuel({ reglage }: { reglage: ReglageMessagerie }) {
  const t = useTranslations("administration.parametres.messagerie");
  const effectif = reglage.effectif;

  return (
    <section className={CARTE}>
      <h2 className={CARTE_TITRE}>{t("actuel")}</h2>
      <Ligne libelle={t("origine")} valeur={t(`source.${effectif.source}`)} />
      <Ligne libelle={t("serveurPort")} valeur={`${effectif.hote}:${effectif.port}`} />
      <Ligne libelle={t("chiffrement.libelle")} valeur={t(`chiffrement.${effectif.chiffrement}`)} />
      <Ligne libelle={t("identifiant")} valeur={effectif.identifiant || ABSENT} />
      <Ligne libelle={t("expediteur")} valeur={effectif.expediteur || ABSENT} />
      <Ligne libelle={t("frontend")} valeur={reglage.frontendUrl} />
      {reglage.frontendUrlLocale && <Alerte type="avertissement">{t("frontendLocale")}</Alerte>}
    </section>
  );
}

function FormulaireMessagerie({ reglage }: { reglage: ReglageMessagerie }) {
  const t = useTranslations("administration.parametres.messagerie");
  const tg = useTranslations("administration");
  const profil = useAdministrateur();
  const cache = useQueryClient();
  const peutAgir = peutParametrerPlateforme(profil);

  // Premier remplissage : le réglage saisi, sinon ce qui sert déjà (sans le mot de passe).
  const depart = reglage.saisi.hote ? reglage.saisi : reglage.effectif;
  const [hote, setHote] = useState(depart.hote);
  const [port, setPort] = useState(String(depart.port));
  const [chiffrement, setChiffrement] = useState<ChiffrementMessagerie>(depart.chiffrement);
  const [identifiant, setIdentifiant] = useState(depart.identifiant);
  const [motDePasse, setMotDePasse] = useState("");
  const [expediteur, setExpediteur] = useState(reglage.saisi.expediteur);
  const [erreurs, setErreurs] = useState<Record<string, string>>({});
  const [erreur, setErreur] = useState<string | null>(null);

  const motDePasseRequis = !reglage.saisi.motDePasseDefini;

  const enregistrement = useMutation({
    mutationFn: () =>
      enregistrerMessagerie({
        hote: hote.trim(),
        port: Number(port),
        chiffrement,
        identifiant: identifiant.trim(),
        motDePasse,
        expediteur: expediteur.trim(),
      }),
    onSuccess: (misAJour) => {
      cache.setQueryData(CLES_ADMINISTRATION.messagerie(), misAJour);
      setErreur(null);
      toast.success(t("succes"));
    },
    onError: (cause) => setErreur(cause instanceof ErreurApi ? cause.message : tg("erreurs.action")),
  });

  const reinitialisation = useMutation({
    mutationFn: reinitialiserMessagerie,
    onSuccess: (misAJour) => {
      cache.setQueryData(CLES_ADMINISTRATION.messagerie(), misAJour);
      toast.success(t("reinitialisee"));
    },
    onError: (cause) => setErreur(cause instanceof ErreurApi ? cause.message : tg("erreurs.action")),
  });

  function valider(): Record<string, string> {
    const trouvees: Record<string, string> = {};
    if (!hote.trim()) trouvees.hote = t("erreurHoteRequis");
    const numero = Number(port);
    if (!Number.isInteger(numero) || numero < 1 || numero > PORT_MAX) trouvees.port = t("erreurPort");
    if (!identifiant.trim()) trouvees.identifiant = t("erreurIdentifiantRequis");
    if (motDePasseRequis && !motDePasse.trim()) trouvees.motDePasse = t("motDePasseRequis");
    return trouvees;
  }

  function soumettre(evenement: FormEvent) {
    evenement.preventDefault();
    const trouvees = valider();
    setErreurs(trouvees);
    if (Object.keys(trouvees).length > 0) return;
    setErreur(null);
    enregistrement.mutate();
  }

  function appliquerGmail() {
    setHote(GMAIL.hote);
    setPort(GMAIL.port);
    setChiffrement(GMAIL.chiffrement);
  }

  const verrouille = !peutAgir || enregistrement.isPending || reinitialisation.isPending;

  return (
    <form noValidate onSubmit={soumettre} className={CARTE}>
      <h2 className={CARTE_TITRE}>{t("formulaire")}</h2>

      <Champ
        libelle={t("hote")}
        aide={t("hoteAide")}
        required
        autoComplete="off"
        value={hote}
        onChange={(evenement) => setHote(evenement.target.value)}
        erreur={erreurs.hote}
        disabled={verrouille}
      />

      <div className="grid gap-x-4 sm:grid-cols-2">
        <Champ
          libelle={t("port")}
          required
          inputMode="numeric"
          value={port}
          onChange={(evenement) => setPort(evenement.target.value)}
          erreur={erreurs.port}
          disabled={verrouille}
        />
        <div className="mb-4 flex flex-col gap-1">
          <Label htmlFor="messagerie-chiffrement" className="text-sm font-semibold text-neutral-800">
            {t("chiffrement.libelle")}
          </Label>
          <Select
            value={chiffrement}
            onValueChange={(valeur) => setChiffrement(valeur as ChiffrementMessagerie)}
            disabled={verrouille}
          >
            <SelectTrigger id="messagerie-chiffrement" className="h-[var(--input-height)] w-full">
              <SelectValue />
            </SelectTrigger>
            <SelectContent>
              {CHIFFREMENTS.map((code) => (
                <SelectItem key={code} value={code}>
                  {t(`chiffrement.${code}`)}
                </SelectItem>
              ))}
            </SelectContent>
          </Select>
        </div>
      </div>

      <Champ
        libelle={t("identifiant")}
        required
        type="email"
        autoComplete="off"
        value={identifiant}
        onChange={(evenement) => setIdentifiant(evenement.target.value)}
        erreur={erreurs.identifiant}
        disabled={verrouille}
      />

      <Champ
        libelle={t("motDePasse")}
        required={motDePasseRequis}
        type="password"
        autoComplete="new-password"
        value={motDePasse}
        onChange={(evenement) => setMotDePasse(evenement.target.value)}
        aide={reglage.saisi.motDePasseDefini ? t("motDePasseDefini") : t("motDePasseAide")}
        erreur={erreurs.motDePasse}
        disabled={verrouille}
      />

      <Champ
        libelle={t("expediteur")}
        aide={t("expediteurAide")}
        autoComplete="off"
        value={expediteur}
        onChange={(evenement) => setExpediteur(evenement.target.value)}
        disabled={verrouille}
      />

      {erreur && <Alerte type="erreur">{erreur}</Alerte>}

      {peutAgir && (
        <div className="mt-2 flex flex-wrap justify-between gap-2">
          <div className="flex flex-wrap gap-2">
            <Bouton variante="ghost" disabled={verrouille} onClick={appliquerGmail}>
              {t("gmail")}
            </Bouton>
            {reglage.saisi.hote && (
              <Bouton
                variante="ghost"
                disabled={verrouille}
                enCours={reinitialisation.isPending}
                onClick={() => reinitialisation.mutate()}
              >
                {t("reinitialiser")}
              </Bouton>
            )}
          </div>
          <Bouton type="submit" variante="primaire" enCours={enregistrement.isPending} disabled={verrouille}>
            {enregistrement.isPending ? t("enCours") : t("enregistrer")}
          </Bouton>
        </div>
      )}
    </form>
  );
}

function TestMessagerie() {
  const t = useTranslations("administration.parametres.messagerie.test");
  const tg = useTranslations("administration");
  const profil = useAdministrateur();
  const peutAgir = peutParametrerPlateforme(profil);

  const [destinataire, setDestinataire] = useState(profil?.email ?? "");
  const [erreurAdresse, setErreurAdresse] = useState<string | undefined>();
  const [resultat, setResultat] = useState<ResultatDiagnosticMessagerie | null>(null);
  const [erreur, setErreur] = useState<string | null>(null);

  const test = useMutation({
    mutationFn: () => testerMessagerie(destinataire.trim()),
    onSuccess: (diagnostic) => {
      setResultat(diagnostic);
      setErreur(null);
    },
    onError: (cause) => {
      setResultat(null);
      setErreur(cause instanceof ErreurApi ? cause.message : tg("erreurs.action"));
    },
  });

  function envoyer(evenement: FormEvent) {
    evenement.preventDefault();
    if (!destinataire.includes("@") || destinataire.trim().length < 5) {
      setErreurAdresse(t("erreurAdresse"));
      return;
    }
    setErreurAdresse(undefined);
    setResultat(null);
    test.mutate();
  }

  return (
    <form noValidate onSubmit={envoyer} className={CARTE}>
      <h2 className={CARTE_TITRE}>{t("titre")}</h2>
      <p className="mb-3 text-sm text-neutral-600">{t("description")}</p>

      <Champ
        libelle={t("destinataire")}
        type="email"
        autoComplete="off"
        value={destinataire}
        onChange={(evenement) => setDestinataire(evenement.target.value)}
        erreur={erreurAdresse}
        disabled={!peutAgir || test.isPending}
      />

      {erreur && <Alerte type="erreur">{erreur}</Alerte>}

      {resultat?.succes && <Alerte type="succes">{t("reussi")}</Alerte>}
      {resultat && !resultat.succes && (
        <Alerte type="erreur">
          <div className="flex flex-col gap-1.5">
            <strong>{resultat.etape === "connexion" ? t("echecConnexion") : t("echecEnvoi")}</strong>
            {resultat.erreur && (
              <span>
                {t("erreurServeur")} : <code className="break-all">{resultat.erreur}</code>
              </span>
            )}
            {resultat.nomEhlo && (
              <span>
                {t("nomEhlo")} : <code className="break-all">{resultat.nomEhlo}</code>
              </span>
            )}
            {resultat.sonde?.etape && (
              <span>
                {t("etapeEnEchec")} : {t(`etape.${resultat.sonde.etape}`)}
                {resultat.sonde.detail && <code className="ml-1 break-all">({resultat.sonde.detail})</code>}
              </span>
            )}
            {resultat.alternatives.map((alternative) => (
              <span key={`${alternative.port}-${alternative.chiffrement}`}>
                {t("alternative", {
                  port: alternative.port,
                  mode: alternative.chiffrement,
                  etat: alternative.ok
                    ? t("alternativeOk")
                    : t("alternativeEchec", { etape: alternative.etape ? t(`etape.${alternative.etape}`) : "" }),
                })}
              </span>
            ))}
            {resultat.conseil && (
              <span>
                {t("conseil")} : {resultat.conseil}
              </span>
            )}
          </div>
        </Alerte>
      )}

      {peutAgir && (
        <div className="mt-2 flex justify-end">
          <Bouton type="submit" variante="primaire" enCours={test.isPending}>
            {test.isPending ? t("enCours") : t("envoyer")}
          </Bouton>
        </div>
      )}
    </form>
  );
}
