"use client";

import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { Trash } from "lucide-react";
import { useRouter } from "next/navigation";
import { useTranslations } from "next-intl";
import { useState } from "react";
import { toast } from "sonner";

import { Alerte, Bouton, Modale } from "@/components/ui";
import { Input } from "@/components/ui/input";
import type { ClientPlateforme } from "@/features/administration";
import {
  lireApercuSuppression,
  supprimerClientDefinitivement,
} from "@/features/administration/adaptateur";
import { ErreurApi } from "@/lib/api";

import { CLES_ADMINISTRATION } from "../../cles";
import { CARTE, CARTE_TITRE } from "../../tons";

/** Les schémas système : ils ne se suppriment jamais, et le serveur les refuse aussi. */
const SCHEMAS_PROTEGES = ["public", "demo"];

/**
 * La zone dangereuse de la fiche : supprimer définitivement une entreprise.
 *
 * **Réservée au superviseur** et masquée pour `public` et `demo` — le serveur
 * refuse dans les deux cas, ces gardes évitent seulement de proposer l'impossible.
 *
 * L'aperçu est lu **à l'ouverture** de la fenêtre : il dit combien de comptes
 * et de chantiers partent, et quelles adresses redeviennent libres. Le bouton
 * ne s'active qu'une fois le nom technique de l'entreprise retapé, pour qu'une
 * suppression ne parte jamais d'un clic distrait.
 */
export function SuppressionClient({
  client,
  peutAgir,
}: {
  client: ClientPlateforme;
  peutAgir: boolean;
}) {
  const t = useTranslations("administration.suppressionDefinitive");
  const cache = useQueryClient();
  const routeur = useRouter();

  const [ouverte, setOuverte] = useState(false);
  const [saisie, setSaisie] = useState("");
  const [erreur, setErreur] = useState<string | null>(null);

  const apercu = useQuery({
    queryKey: [...CLES_ADMINISTRATION.client(client.id), "suppression"],
    queryFn: () => lireApercuSuppression(client.id),
    enabled: ouverte,
  });

  const suppression = useMutation({
    mutationFn: () => supprimerClientDefinitivement(client.id, saisie.trim()),
    onSuccess: () => {
      toast.success(t("succes", { client: client.nomCommercial }));
      cache.removeQueries({ queryKey: CLES_ADMINISTRATION.client(client.id) });
      void cache.invalidateQueries({ queryKey: CLES_ADMINISTRATION.clients() });
      void cache.invalidateQueries({ queryKey: CLES_ADMINISTRATION.indicateurs() });
      routeur.replace("/admin/clients");
    },
    onError: (cause) => setErreur(cause instanceof ErreurApi ? cause.message : t("erreur")),
  });

  if (SCHEMAS_PROTEGES.includes(client.slug.toLowerCase())) return null;
  // Le support consulte, il ne supprime pas : pas de zone dangereuse à lui montrer.
  if (!peutAgir) return null;

  function fermer() {
    if (suppression.isPending) return;
    setOuverte(false);
    setSaisie("");
    setErreur(null);
  }

  const donnees = apercu.data;
  const desactivee = donnees !== undefined && !donnees.suppressionActivee;
  const confirmationValide = saisie.trim().toLowerCase() === client.slug.toLowerCase();
  const peutConfirmer = donnees !== undefined && !desactivee && confirmationValide;

  return (
    <>
      <section className={`${CARTE} border-erreur lg:col-span-3`}>
        <h2 className={CARTE_TITRE}>{t("titre")}</h2>
        <p className="text-sm text-neutral-600">{t("description")}</p>
        <div>
          <Bouton
            variante="danger"
            taille="sm"
            iconeGauche={<Trash size={16} />}
            onClick={() => setOuverte(true)}
          >
            {t("ouvrir")}
          </Bouton>
        </div>
      </section>

      <Modale
        ouverte={ouverte}
        titre={t("titreModale", { client: client.nomCommercial })}
        onFermer={suppression.isPending ? undefined : fermer}
        actions={
          <>
            <Bouton variante="ghost" onClick={fermer} disabled={suppression.isPending}>
              {t("annuler")}
            </Bouton>
            <Bouton
              variante="danger"
              disabled={!peutConfirmer}
              enCours={suppression.isPending}
              onClick={() => {
                setErreur(null);
                suppression.mutate();
              }}
            >
              {suppression.isPending ? t("enCours") : t("confirmer")}
            </Bouton>
          </>
        }
      >
        <div className="flex flex-col gap-4">
          <Alerte type="erreur">{t("avertissement")}</Alerte>

          {apercu.isPending && <p className="text-sm text-neutral-600">{t("chargementApercu")}</p>}
          {apercu.isError && <Alerte type="erreur">{t("erreurApercu")}</Alerte>}
          {desactivee && <Alerte type="avertissement">{t("desactivee")}</Alerte>}

          {donnees && (
            <>
              <ul className="flex list-disc flex-col gap-1 pl-5 text-sm text-neutral-700">
                <li>{t("compteUtilisateurs", { nombre: donnees.nbUtilisateurs })}</li>
                <li>{t("compteProjets", { nombre: donnees.nbProjets })}</li>
                {donnees.nbPaiements > 0 && (
                  <li>{t("comptePaiements", { nombre: donnees.nbPaiements })}</li>
                )}
              </ul>

              <div className="flex flex-col gap-1.5">
                <p className="text-sm font-medium text-neutral-800">
                  {t("adressesLiberees", { nombre: donnees.adresses.length })}
                </p>
                <ul className="flex max-h-32 flex-col gap-0.5 overflow-y-auto rounded-md border border-neutral-200 bg-neutral-50 px-3 py-2 text-sm text-neutral-700">
                  {donnees.adresses.map((adresse) => (
                    <li key={adresse}>{adresse}</li>
                  ))}
                </ul>
              </div>
            </>
          )}

          <label className="flex flex-col gap-1.5 text-sm font-medium text-neutral-800">
            {t("saisie", { slug: client.slug })}
            <Input
              value={saisie}
              onChange={(evenement) => setSaisie(evenement.target.value)}
              placeholder={client.slug}
              autoComplete="off"
              disabled={suppression.isPending}
            />
          </label>

          {erreur && <Alerte type="erreur">{erreur}</Alerte>}
        </div>
      </Modale>
    </>
  );
}
