"use client";

import { useMutation, useQueryClient } from "@tanstack/react-query";
import { useTranslations } from "next-intl";
import { toast } from "sonner";

import { Checkbox } from "@/components/ui/checkbox";
import type { ClientPlateforme } from "@/features/administration";
import { basculerModuleClient } from "@/features/administration/adaptateur";
import { ErreurApi } from "@/lib/api";

import { CLES_ADMINISTRATION } from "../../cles";
import { CARTE, CARTE_TITRE } from "../../tons";

/**
 * Les modules de l'entreprise : cocher en active un, décocher le retire de son
 * espace (ses réglages sont conservés, A-10). Le serveur prévient le DG par e-mail.
 */
export function ModulesClient({ client, peutAgir }: { client: ClientPlateforme; peutAgir: boolean }) {
  const t = useTranslations("administration.fiche");
  const cache = useQueryClient();
  // « Administration » n'est pas un module qu'on souscrit : tout client l'a, et le retirer n'aurait aucun sens.
  const modules = client.modules.filter((module) => module.code.toLowerCase() !== "administration");

  const bascule = useMutation({
    mutationFn: ({ moduleId, actif }: { moduleId: string; actif: boolean }) =>
      basculerModuleClient(client.id, moduleId, actif),
    onSuccess: (misAJour) => {
      cache.setQueryData(CLES_ADMINISTRATION.client(client.id), misAJour);
      toast.success(t("modulesMisAJour"));
    },
    onError: (cause) => toast.error(cause instanceof ErreurApi ? cause.message : t("modulesErreur")),
  });

  return (
    <section className={CARTE}>
      <h2 className={CARTE_TITRE}>{t("modules")}</h2>
      {modules.length === 0 ? (
        <p className="text-sm text-neutral-500">{t("modulesAucun")}</p>
      ) : (
        <ul className="flex flex-col gap-2">
          {modules.map((module) => (
            <li key={module.id} className="flex items-center gap-2">
              <Checkbox
                id={`module-client-${module.id}`}
                checked={module.actif}
                disabled={!peutAgir || bascule.isPending}
                onCheckedChange={(coche) => bascule.mutate({ moduleId: module.id, actif: coche === true })}
              />
              <label htmlFor={`module-client-${module.id}`} className="text-sm text-neutral-900">
                {module.libelle}
              </label>
            </li>
          ))}
        </ul>
      )}
    </section>
  );
}
