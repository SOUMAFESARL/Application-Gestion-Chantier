"use client";

import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { useTranslations } from "next-intl";
import { toast } from "sonner";

import { EtatChargement, EtatErreur } from "@/components/ui";
import { Checkbox } from "@/components/ui/checkbox";
import { basculerPermission, listerPermissions } from "@/features/administration/adaptateur";
import { peutParametrerPlateforme } from "@/features/administration/regles";

import { useAdministrateur } from "../../ContexteAdministrateur";

const CLE_PERMISSIONS = ["administration", "permissions"] as const;

/**
 * Le catalogue des permissions, en lecture : elles viennent du code du serveur.
 * La seule action est d'en désactiver une — elle disparaît alors de tous les
 * rôles de toutes les entreprises, et se réactive de la même façon.
 */
export function ListePermissions() {
  const t = useTranslations("administration.parametres.modules");
  const profil = useAdministrateur();
  const cache = useQueryClient();
  const peutAgir = peutParametrerPlateforme(profil);

  const requete = useQuery({
    queryKey: CLE_PERMISSIONS,
    queryFn: ({ signal }) => listerPermissions(signal),
  });

  const bascule = useMutation({
    mutationFn: ({ id, actif }: { id: string; actif: boolean }) => basculerPermission(id, actif),
    onSuccess: (permission) => {
      toast.success(t(permission.actif ? "cataloguePermissionsActivee" : "cataloguePermissionsDesactivee"));
      void cache.invalidateQueries({ queryKey: CLE_PERMISSIONS });
    },
    onError: (cause) => toast.error(cause instanceof Error ? cause.message : ""),
  });

  if (requete.isPending) return <EtatChargement message={t("cataloguePermissionsTitre")} />;
  if (requete.isError) return <EtatErreur message={t("cataloguePermissionsErreur")} onReessayer={() => requete.refetch()} />;

  const permissions = [...requete.data].sort((a, b) => a.code.localeCompare(b.code));

  return (
    <section className="flex flex-col gap-3">
      <div>
        <h2 className="text-base font-semibold text-neutral-900">{t("cataloguePermissionsTitre")}</h2>
        <p className="text-sm text-neutral-600">{t("cataloguePermissionsDescription")}</p>
      </div>
      <div className="overflow-x-auto rounded-md border border-neutral-200 bg-neutral-0">
        <table className="w-full border-collapse text-sm">
          <thead className="bg-neutral-50 text-left text-neutral-700">
            <tr>
              <th className="px-3 py-2 font-semibold">{t("cataloguePermissionsColonneCode")}</th>
              <th className="px-3 py-2 font-semibold">{t("cataloguePermissionsColonneLibelle")}</th>
              <th className="px-3 py-2 text-center font-semibold">{t("cataloguePermissionsColonneActive")}</th>
            </tr>
          </thead>
          <tbody>
            {permissions.map((permission) => (
              <tr key={permission.id} className="border-t border-neutral-200">
                <td className="px-3 py-2 font-mono text-xs text-neutral-700">{permission.code}</td>
                <td className="px-3 py-2">
                  <span className="text-neutral-900">{permission.libelle}</span>
                  {permission.description && (
                    <span className="block text-xs text-neutral-500">{permission.description}</span>
                  )}
                </td>
                <td className="px-3 py-2 text-center">
                  <Checkbox
                    checked={permission.actif}
                    disabled={!peutAgir || bascule.isPending}
                    onCheckedChange={(coche) => bascule.mutate({ id: permission.id, actif: coche === true })}
                    aria-label={permission.libelle}
                  />
                </td>
              </tr>
            ))}
          </tbody>
        </table>
      </div>
    </section>
  );
}
